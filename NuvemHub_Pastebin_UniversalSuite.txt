local CACHE_FOLDER = "NuvemHubCache"

local HasFS =
    type(isfile) == "function"
    and type(writefile) == "function"
    and type(readfile) == "function"
    and type(makefolder) == "function"

local function ensureFolder()
    if not HasFS then
        return
    end

    if type(isfolder) == "function" then
        if not isfolder(CACHE_FOLDER) then
            makefolder(CACHE_FOLDER)
        end
    else
        pcall(makefolder, CACHE_FOLDER)
    end
end

local function fetchCached(name, remote)
    if not HasFS then
        return game:HttpGet(REPO .. remote)
    end

    ensureFolder()

    local path = CACHE_FOLDER .. "/" .. name

    if isfile(path) then
        local ok, source = pcall(readfile, path)
        if ok and type(source) == "string" and #source > 100 then
            return source
        end
    end

    local source = game:HttpGet(REPO .. remote)
    writefile(path, source)

    return source
end

local function loadCached(name, remote)
    local source = fetchCached(name, remote)
    local fn, err = loadstring(source)
    assert(fn, err)
    return fn()
end

local function round(n)
    return math.floor(n + 0.5)
end

local function clamp(n, a, b)
    return math.clamp(n, a, b)
end

local function lower(v)
    return string.lower(tostring(v or ""))
end

local function containsAny(text, words)
    text = lower(text)

    for _, word in ipairs(words) do
        if string.find(text, lower(word), 1, true) then
            return true
        end
    end

    return false
end

local function normalizeCompact(text)
    text = lower(text)
    return string.gsub(text, "[^%w]", "")
end

local function tokenize(text)
    text = lower(text)
    text = string.gsub(text, "[^%w%s]", " ")

    local ignored = {
        the = true,
        a = true,
        an = true,
        item = true,
        required = true,
        requires = true,
        need = true,
        needs = true,
        use = true,
        door = true,
    }

    local tokens = {}

    for token in string.gmatch(text, "%S+") do
        if token == "card" or token == "accesscard" then
            token = "keycard"
        end

        if not ignored[token] and #token > 1 then
            tokens[token] = true
        end
    end

    local compact = normalizeCompact(text)

    if string.find(compact, "keycard", 1, true)
        or string.find(compact, "accesscard", 1, true) then
        tokens.keycard = true
    end

    for _, color in ipairs({
        "green", "red", "blue", "yellow", "orange",
        "purple", "pink", "white", "black",
        "gold", "silver", "cyan",
    }) do
        if string.find(compact, color, 1, true) then
            tokens[color] = true
        end
    end

    return tokens
end

local function tokenSimilarity(a, b)
    local A = tokenize(a)
    local B = tokenize(b)

    local intersection = 0
    local union = {}

    for token in pairs(A) do
        union[token] = true
        if B[token] then
            intersection += 1
        end
    end

    for token in pairs(B) do
        union[token] = true
    end

    local total = 0
    for _ in pairs(union) do
        total += 1
    end

    if total == 0 then
        return 0
    end

    return intersection / total
end

local function nameMatchScore(a, b)
    if not a or not b then
        return 0
    end

    local A = normalizeCompact(a)
    local B = normalizeCompact(b)

    if A == "" or B == "" then
        return 0
    end

    if A == B then
        return 100
    end

    local score = tokenSimilarity(a, b) * 100

    if string.find(A, B, 1, true)
        or string.find(B, A, 1, true) then
        score = math.max(score, 75)
    end

    if string.find(A, "keycard", 1, true)
        and string.find(B, "keycard", 1, true) then
        score += 15
    end

    return clamp(score, 0, 100)
end

function Character:GetHumanoid()
    local character = LocalPlayer.Character
    return character and character:FindFirstChildOfClass("Humanoid") or nil
end

function Character:GetRoot()
    local character = LocalPlayer.Character

    if not character then
        return nil
    end

    return character:FindFirstChild("HumanoidRootPart")
        or character.PrimaryPart
end

function Character:CaptureOriginal()
    local humanoid = self:GetHumanoid()

    if not humanoid then
        return
    end

    self.Original.Speed = humanoid.WalkSpeed
    self.Original.JumpPower = humanoid.JumpPower
    self.Original.UseJumpPower = humanoid.UseJumpPower
end

function Character:SetSpeed(value)
    State.Player.Speed = value

    local humanoid = self:GetHumanoid()

    if humanoid then
        humanoid.WalkSpeed = value
    end
end

function Character:SetJump(value)
    State.Player.JumpPower = value

    local humanoid = self:GetHumanoid()

    if humanoid then
        humanoid.UseJumpPower = true
        humanoid.JumpPower = value
    end
end

function Character:Apply()
    self:SetSpeed(State.Player.Speed)
    self:SetJump(State.Player.JumpPower)
end

function Character:Restore()
    local humanoid = self:GetHumanoid()

    if not humanoid then
        return
    end

    if self.Original.Speed ~= nil then
        humanoid.WalkSpeed = self.Original.Speed
    end

    if self.Original.JumpPower ~= nil then
        humanoid.JumpPower = self.Original.JumpPower
    end

    if self.Original.UseJumpPower ~= nil then
        humanoid.UseJumpPower = self.Original.UseJumpPower
    end
end

local OriginalCollisions = {}

local function restoreCollision()
    for part, value in pairs(OriginalCollisions) do
        if part and part.Parent then
            part.CanCollide = value
        end
    end

    table.clear(OriginalCollisions)
end

connect(RunService.Heartbeat, function()
    if not State.Running or not State.Player.Noclip then
        return
    end

    local character = LocalPlayer.Character

    if not character then
        return
    end

    for _, object in ipairs(character:GetDescendants()) do
        if object:IsA("BasePart") then
            if OriginalCollisions[object] == nil then
                OriginalCollisions[object] = object.CanCollide
            end

            object.CanCollide = false
        end
    end
end)

connect(UIS.JumpRequest, function()
    if not State.Running or not State.Player.InfiniteJump then
        return
    end

    local humanoid = Character:GetHumanoid()

    if humanoid then
        humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
    end
end)

local HostileWords = {
    "monster", "enemy", "killer", "creature",
    "zombie", "boss", "hostile", "demon",
    "mutant", "beast", "hunter", "infected",
}

local function isPlayerRelatedModel(model)
    if not model or not model:IsA("Model") then
        return false
    end

    for _, player in ipairs(Players:GetPlayers()) do
        local character = player.Character

        if character and (
            model == character
            or model:IsDescendantOf(character)
        ) then
            return true
        end
    end

    return false
end

local function getEntityRoot(model)
    if not model or not model:IsA("Model") then
        return nil
    end

    return model:FindFirstChild("HumanoidRootPart", true)
        or model.PrimaryPart
        or model:FindFirstChildWhichIsA("BasePart", true)
end

local function getEntityHumanoid(model)
    if not model or not model:IsA("Model") then
        return nil
    end

    return model:FindFirstChildWhichIsA("Humanoid", true)
end

local function modelOrAncestorsHaveTag(model, tags)
    local current = model

    for _ = 1, 5 do
        if not current or current == Workspace then
            break
        end

        if current:IsA("Model") then
            for _, tag in ipairs(tags) do
                if CollectionService:HasTag(current, tag) then
                    return true, current
                end
            end
        end

        current = current.Parent
    end

    return false, nil
end

local function modelOrAncestorsHaveTrueAttribute(model, names)
    local current = model

    for _ = 1, 5 do
        if not current or current == Workspace then
            break
        end

        if current:IsA("Model") then
            for _, name in ipairs(names) do
                if current:GetAttribute(name) == true then
                    return true, current
                end
            end
        end

        current = current.Parent
    end

    return false, nil
end

local function modelAncestryBlob(model)
    local pieces = {}
    local current = model

    for _ = 1, 5 do
        if not current or current == Workspace then
            break
        end

        if current:IsA("Model") then
            table.insert(pieces, current.Name)
        end

        current = current.Parent
    end

    return lower(table.concat(pieces, " "))
end

local function resolveEntityModel(source)
    if not source then
        return nil
    end

    local model = source:IsA("Model")
        and source
        or source:FindFirstAncestorOfClass("Model")

    if not model or isPlayerRelatedModel(model) then
        return nil
    end

    local candidates = {}
    local current = model

    for _ = 1, 5 do
        if not current or current == Workspace then
            break
        end

        if current:IsA("Model") then
            table.insert(candidates, current)
        end

        current = current.Parent
    end

    -- Prefer an ancestor that carries an explicit entity signal.
    for _, candidate in ipairs(candidates) do
        local tagged = modelOrAncestorsHaveTag(candidate, {
            "Monster", "Enemy", "NPC", "Entity",
        })

        local attributed = modelOrAncestorsHaveTrueAttribute(candidate, {
            "Hostile", "Enemy", "NPC", "Entity",
        })

        if tagged or attributed then
            return candidate
        end
    end

    -- Otherwise prefer the nearest rig-like model.
    for _, candidate in ipairs(candidates) do
        if getEntityHumanoid(candidate)
            or candidate:FindFirstChildWhichIsA("AnimationController", true) then
            return candidate
        end
    end

    return model
end

local function classifyEntity(model)
    if not model
        or not model:IsA("Model")
        or isPlayerRelatedModel(model) then
        return nil
    end

    local humanoid = getEntityHumanoid(model)
    local ancestryBlob = modelAncestryBlob(model)

    local monsterTag = modelOrAncestorsHaveTag(
        model,
        { "Monster", "Enemy" }
    )

    local monsterAttribute = modelOrAncestorsHaveTrueAttribute(
        model,
        { "Hostile", "Enemy", "IsMonster" }
    )

    if humanoid and (
        monsterTag
        or monsterAttribute
        or containsAny(ancestryBlob, HostileWords)
    ) then
        return "MONSTER"
    end

    local npcTag = modelOrAncestorsHaveTag(
        model,
        { "NPC" }
    )

    local npcAttribute = modelOrAncestorsHaveTrueAttribute(
        model,
        { "NPC", "IsNPC" }
    )

    if humanoid and (npcTag or npcAttribute) then
        return "NPC"
    end

    if humanoid then
        return "NPC"
    end

    local entityTag = modelOrAncestorsHaveTag(
        model,
        { "Entity" }
    )

    if entityTag
        or model:FindFirstChildWhichIsA("AnimationController", true) then
        return "ENTITY"
    end

    return nil
end

function EntityESP:ShouldShow(class)
    if class == "PLAYER" then
        return State.EntityESP.Players
    elseif class == "NPC" then
        return State.EntityESP.NPCs
    elseif class == "MONSTER" then
        return State.EntityESP.Monsters
    elseif class == "ENTITY" then
        return State.EntityESP.Entities
    end

    return false
end

    if class == "MONSTER" then
        return State.EntityESP.MonsterMaxVisible
    elseif class == "NPC" then
        return State.EntityESP.NPCMaxVisible
    elseif class == "PLAYER" then
        return State.EntityESP.PlayerMaxVisible
    end

    return State.EntityESP.EntityMaxVisible
end

function EntityESP:GetCounts()
    local counts = {
        PLAYER = 0,
        NPC = 0,
        MONSTER = 0,
        ENTITY = 0,
    }

    for _, data in pairs(self.Objects) do
        if counts[data.Class] ~= nil then
            counts[data.Class] += 1
        end
    end

    return counts
end

function EntityESP:GetCount()
    local total = 0

    for _ in pairs(self.Objects) do
        total += 1
    end

    return total
end

function EntityESP:DestroyVisual(data)
    if not data or not data.Visual then
        return
    end

    pcall(function()
        data.Visual.Highlight:Destroy()
    end)

    pcall(function()
        data.Visual.Gui:Destroy()
    end)

    data.Visual = nil
end

function EntityESP:MaybeAlert(data, isNew)
    if not isNew
        or data.Class ~= "MONSTER"
        or not State.EntityESP.MonsterAlerts
        or self.Alerted[data.Model] then
        return
    end

    local playerRoot = Character:GetRoot()
    local root = data.Root

    if not playerRoot or not root then
        return
    end

    local distance = (playerRoot.Position - root.Position).Magnitude

    if distance <= State.EntityESP.MonsterAlertDistance then
        self.Alerted[data.Model] = true

        card(
            "WARNING",
            "Monster detected",
            data.Model.Name
                .. " | "
                .. tostring(round(distance))
                .. " studs",
            5
        )
    end
end

function EntityESP:AddOrUpdate(source)
    local model = resolveEntityModel(source)

    if not model or not model.Parent then
        return false
    end

    local class = classifyEntity(model)
    local root = getEntityRoot(model)

    if not class or not root then
        return false
    end

    local data = self.Objects[model]
    local isNew = data == nil

    if not data then
        data = {
            Model = model,
            Class = class,
            Root = root,
            Visual = nil,
            LastVisible = 0,
            LastClassCheck = 0,
        }

        self.Objects[model] = data
    else
        local classChanged = data.Class ~= class
        local rootChanged = data.Root ~= root

        if classChanged then
            self:DestroyVisual(data)
            data.Class = class
        end

        if rootChanged then
            data.Root = root

            if data.Visual then
                data.Visual.Gui.Adornee = root
            end
        end
    end

    data.LastClassCheck = os.clock()
    self:MaybeAlert(data, isNew)

    return true
end

function EntityESP:Remove(model)
    local data = self.Objects[model]

    if not data then
        return
    end

    self:DestroyVisual(data)
    self.Objects[model] = nil
end

function EntityESP:Clear()
    local remove = {}

    for model in pairs(self.Objects) do
        table.insert(remove, model)
    end

    for _, model in ipairs(remove) do
        self:Remove(model)
    end
end

function EntityESP:UpdateVisual(data, distance)
    local visual = self:EnsureVisual(data)

    if not visual then
        return
    end

    local style = self.Styles[data.Class] or self.Styles.ENTITY
    local nameText = (State.Privacy.StreamerMode and data.Class == "PLAYER") and "PLAYER" or data.Model.Name

    if State.EntityESP.ShowDistance then
        nameText ..= " | " .. tostring(round(distance)) .. "st"
    end

    visual.Name.Text = nameText
    visual.Tag.Text = style.Tag

    local humanoid = getEntityHumanoid(data.Model)
    local showHealth = false

    if data.Class == "MONSTER" then
        showHealth = State.EntityESP.MonsterHealthBar
    elseif data.Class == "NPC" then
        showHealth = State.EntityESP.NPCHealthBar
    end

    if showHealth and humanoid then
        local maxHealth = math.max(humanoid.MaxHealth, 1)
        local ratio = clamp(humanoid.Health / maxHealth, 0, 1)

        visual.HealthBG.Visible = true
        visual.HealthFill.Size = UDim2.new(ratio, 0, 1, 0)
        visual.HealthFill.BackgroundColor3 = style.Color
    else
        visual.HealthBG.Visible = false
    end
end

local EntityTracker = {
    RetryToken = setmetatable({}, { __mode = "k" }),
}

function EntityTracker:Queue(source)
    local model = resolveEntityModel(source)

    if not model or not model.Parent then
        return
    end

    local token = (self.RetryToken[model] or 0) + 1
    self.RetryToken[model] = token

    if EntityESP:AddOrUpdate(model) then
        return
    end

    for _, delayTime in ipairs({ 0.12, 0.35, 0.8, 1.8, 3.5, 6.0 }) do
        task.delay(delayTime, function()
            if not State.Running
                or not model.Parent
                or self.RetryToken[model] ~= token then
                return
            end

            local resolved = resolveEntityModel(model) or model

            if EntityESP:AddOrUpdate(resolved) then
                self.RetryToken[model] = nil
            end
        end)
    end
end

function EntityTracker:ScanNearby()
    local root = Character:GetRoot()

    if not root then
        return
    end

    local radius = math.max(
        State.EntityESP.MaxDistance + 250,
        State.EntityESP.RecoveryRadius
    )

    local params = OverlapParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {
        LocalPlayer.Character,
    }
    params.MaxParts = 3000

    local parts = Workspace:GetPartBoundsInRadius(
        root.Position,
        radius,
        params
    )

    local seen = {}

    for _, part in ipairs(parts) do
        local model = part:FindFirstAncestorOfClass("Model")

        if model and not seen[model] then
            seen[model] = true
            self:Queue(model)
        end
    end

    for _, tag in ipairs({ "Monster", "Enemy", "NPC", "Entity" }) do
        for _, instance in ipairs(CollectionService:GetTagged(tag)) do
            self:Queue(instance)
        end
    end

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local character = player.Character

            if not EntityESP.Objects[character] then
                local rootPart = getEntityRoot(character)

                if rootPart then
                    EntityESP.Objects[character] = {
                        Model = character,
                        Class = "PLAYER",
                        Root = rootPart,
                        Visual = nil,
                        LastVisible = 0,
                        LastClassCheck = os.clock(),
                    }
                end
            end
        end
    end
end

function EntityTracker:FullScan()
    local descendants = Workspace:GetDescendants()

    for index, object in ipairs(descendants) do
        if object:IsA("Model")
            or object:IsA("Humanoid")
            or object:IsA("AnimationController") then
            self:Queue(object)
        end

        if index % 950 == 0 then
            RunService.Heartbeat:Wait()
        end
    end
end

function Scanner:IsPlayerCharacter(object)
    local model = object

    if not model:IsA("Model") then
        model = object:FindFirstAncestorOfClass("Model")
    end

    return model and Players:GetPlayerFromCharacter(model) ~= nil
end

function Scanner:GetPrompt(candidate)
    return candidate
        and candidate:FindFirstChildWhichIsA("ProximityPrompt", true)
        or nil
end

function Scanner:GetClickDetector(candidate)
    return candidate
        and candidate:FindFirstChildWhichIsA("ClickDetector", true)
        or nil
end

function Scanner:GetNameBlob(candidate)
    local pieces = { candidate.Name }

    local parent = candidate.Parent

    for _ = 1, 2 do
        if parent and parent ~= Workspace then
            table.insert(pieces, parent.Name)
            parent = parent.Parent
        end
    end

    local prompt = self:GetPrompt(candidate)

    if prompt then
        table.insert(pieces, tostring(prompt.ActionText))
        table.insert(pieces, tostring(prompt.ObjectText))
    end

    return lower(table.concat(pieces, " "))
end

function Scanner:IsMeaningfulOwnName(object)
    local name = lower(object.Name)

    return containsAny(name, self.ItemWords)
        or containsAny(name, self.DoorWords)
        or containsAny(name, self.SecretWords)
        or containsAny(name, self.CodeWords)
        or containsAny(name, self.TerminalWords)
        or containsAny(name, self.SwitchWords)
end

function Scanner:GetCandidate(object)
    if not object or self:IsPlayerCharacter(object) then
        return nil
    end

    if object:IsA("Tool") then
        return object
    end

    if object:IsA("ProximityPrompt")
        or object:IsA("ClickDetector") then

        local tool = object:FindFirstAncestorOfClass("Tool")

        if tool then
            return tool
        end

        local model = object:FindFirstAncestorOfClass("Model")

        if model and not self:IsPlayerCharacter(model) then
            return model
        end

        return object.Parent
    end

    if object:IsA("BasePart") then
        local tool = object:FindFirstAncestorOfClass("Tool")

        if tool then
            return tool
        end

        -- critical fix for names such as "Green Keycard"
        if self:IsMeaningfulOwnName(object) then
            return object
        end

        local model = object:FindFirstAncestorOfClass("Model")

        if model and not self:IsPlayerCharacter(model) then
            return model
        end

        return object
    end

    if object:IsA("Model")
        and not self:IsPlayerCharacter(object) then
        return object
    end

    return nil
end

function Scanner:GetAnchor(candidate)
    if not candidate then
        return nil
    end

    if candidate:IsA("BasePart") then
        return candidate
    end

    if candidate:IsA("Tool") then
        return candidate:FindFirstChild("Handle")
            or candidate:FindFirstChildWhichIsA("BasePart", true)
    end

    if candidate:IsA("Model") then
        return candidate:FindFirstChild("HumanoidRootPart")
            or candidate.PrimaryPart
            or candidate:FindFirstChildWhichIsA("BasePart", true)
    end

    return nil
end

function Scanner:GetAdornee(candidate)
    if candidate:IsA("Model")
        or candidate:IsA("BasePart") then
        return candidate
    end

    return self:GetAnchor(candidate)
end

function Scanner:HasAnyTag(candidate, tags)
    for _, tag in ipairs(tags) do
        if CollectionService:HasTag(candidate, tag) then
            return true
        end
    end

    return false
end

function Scanner:GetAttributeValue(candidate, names)
    for _, name in ipairs(names) do
        local value = candidate:GetAttribute(name)

        if value ~= nil then
            return value
        end
    end

    return nil
end

function Scanner:FindValue(candidate, words)
    for _, object in ipairs(candidate:GetDescendants()) do
        if object:IsA("StringValue")
            or object:IsA("IntValue")
            or object:IsA("NumberValue")
            or object:IsA("ObjectValue") then

            local objectName = lower(object.Name)

            for _, word in ipairs(words) do
                if string.find(
                    objectName,
                    lower(word),
                    1,
                    true
                ) then

                    if object:IsA("ObjectValue") then
                        if object.Value then
                            return object.Value.Name
                        end
                    else
                        return object.Value
                    end
                end
            end
        end
    end

    return nil
end

function Scanner:IsLocked(candidate)
    for _, name in ipairs({
        "Locked",
        "IsLocked",
        "RequiresKey",
        "NeedsKey",
        "RequiresAccess",
    }) do
        if candidate:GetAttribute(name) == true then
            return true
        end
    end

    for _, object in ipairs(candidate:GetDescendants()) do
        if object:IsA("BoolValue")
            and string.find(
                lower(object.Name),
                "locked",
                1,
                true
            )
            and object.Value == true then
            return true
        end
    end

    return false
end

function Scanner:GetAccess(candidate)
    local access = {
        Type = "UNKNOWN",
        Item = nil,
        Code = nil,
        Trigger = nil,
        Locked = self:IsLocked(candidate),
    }

    local types = {}

    local item =
        self:GetAttributeValue(
            candidate,
            {
                "RequiredItem",
                "RequiredKey",
                "KeyName",
                "ItemRequired",
                "RequiredTool",
                "AccessItem",
            }
        )

    if item ~= nil and tostring(item) ~= "" then
        access.Item = tostring(item)
        types.ITEM = true
    else
        local value =
            self:FindValue(
                candidate,
                {
                    "requireditem",
                    "requiredkey",
                    "keyname",
                    "itemrequired",
                    "requiredtool",
                    "accessitem",
                }
            )

        if value ~= nil and tostring(value) ~= "" then
            access.Item = tostring(value)
            types.ITEM = true
        end
    end

    local code =
        self:GetAttributeValue(
            candidate,
            {
                "RequiredCode",
                "AccessCode",
                "Password",
                "Passcode",
                "PIN",
                "Pin",
                "Code",
                "Combination",
            }
        )

    if code ~= nil and tostring(code) ~= "" then
        access.Code = tostring(code)
        types.CODE = true
    else
        local value =
            self:FindValue(
                candidate,
                {
                    "requiredcode",
                    "accesscode",
                    "password",
                    "passcode",
                    "pin",
                    "combination",
                }
            )

        if value ~= nil and tostring(value) ~= "" then
            access.Code = tostring(value)
            types.CODE = true
        end
    end

    local trigger =
        self:GetAttributeValue(
            candidate,
            {
                "RequiredSwitch",
                "RequiredLever",
                "RequiredGenerator",
                "RequiredButton",
                "RequiredTrigger",
            }
        )

    if trigger ~= nil and tostring(trigger) ~= "" then
        access.Trigger = tostring(trigger)
        types.SWITCH = true
    end

    local prompt = self:GetPrompt(candidate)

    if prompt then
        local promptText =
            lower(
                tostring(prompt.ActionText)
                .. " "
                .. tostring(prompt.ObjectText)
            )

        if containsAny(promptText, self.CodeWords) then
            types.CODE = true
        end

        if containsAny(promptText, self.SwitchWords) then
            types.SWITCH = true
        end
    end

    local blob = self:GetNameBlob(candidate)

    if containsAny(blob, self.CodeWords) then
        types.CODE = true
    end

    if containsAny(blob, self.SwitchWords)
        and not containsAny(blob, self.HighValueItemWords) then
        types.SWITCH = true
    end

    local count = 0
    for _ in pairs(types) do
        count += 1
    end

    if count > 1 then
        access.Type = "MULTI"
    elseif types.ITEM then
        access.Type = "ITEM"
    elseif types.CODE then
        access.Type = "CODE"
    elseif types.SWITCH then
        access.Type = "SWITCH"
    elseif not access.Locked then
        access.Type = "FREE"
    end

    return access
end

function Scanner:Classify(candidate)
    if not candidate
        or not candidate:IsDescendantOf(Workspace)
        or self:IsPlayerCharacter(candidate) then
        return nil
    end

    local anchor = self:GetAnchor(candidate)

    if not anchor then
        return nil
    end

    local blob = self:GetNameBlob(candidate)
    local prompt = self:GetPrompt(candidate)
    local clickDetector = self:GetClickDetector(candidate)
    local access = self:GetAccess(candidate)

    local score = {
        ITEM = 0,
        DOOR = 0,
        SECRET_DOOR = 0,
        KEYPAD = 0,
        TERMINAL = 0,
        SWITCH = 0,
    }

    local strongDoor = false
    local strongItem = false

    if candidate:IsA("Tool") then
        score.ITEM += 115
        strongItem = true
    end

    if self:HasAnyTag(
        candidate,
        {
            "Item",
            "Pickup",
            "Loot",
            "Key",
            "Collectible",
        }
    ) then
        score.ITEM += 115
        strongItem = true
    end

    if self:HasAnyTag(
        candidate,
        {
            "Door",
            "Gate",
            "VaultDoor",
        }
    ) then
        score.DOOR += 120
        strongDoor = true
    end

    if self:HasAnyTag(
        candidate,
        {
            "SecretDoor",
            "HiddenDoor",
            "SecretEntrance",
        }
    ) then
        score.SECRET_DOOR += 145
        strongDoor = true
    end

    if self:HasAnyTag(
        candidate,
        {
            "Keypad",
            "CodePanel",
        }
    ) then
        score.KEYPAD += 125
    end

    if self:HasAnyTag(
        candidate,
        {
            "Terminal",
            "AccessPanel",
            "Console",
        }
    ) then
        score.TERMINAL += 125
    end

    if self:HasAnyTag(
        candidate,
        {
            "Switch",
            "Lever",
            "Generator",
            "Button",
        }
    ) then
        score.SWITCH += 125
    end

    if candidate:GetAttribute("Item") == true
        or candidate:GetAttribute("Pickup") == true
        or candidate:GetAttribute("Collectible") == true then
        score.ITEM += 90
        strongItem = true
    end

    if candidate:GetAttribute("Door") == true then
        score.DOOR += 95
        strongDoor = true
    end

    if candidate:GetAttribute("Secret") == true
        or candidate:GetAttribute("Hidden") == true then
        score.SECRET_DOOR += 100
        strongDoor = true
    end

    if containsAny(blob, self.ItemWords) then
        score.ITEM += 45
    end

    if containsAny(blob, self.HighValueItemWords) then
        score.ITEM += 65
        strongItem = true
    end

    if containsAny(blob, self.DoorWords) then
        score.DOOR +=
            State.Interactables.DoorMode == "All"
            and 70
            or 24
    end

    if containsAny(blob, self.SecretWords) then
        score.SECRET_DOOR += 55
    end

    if containsAny(blob, self.CodeWords) then
        score.KEYPAD += 58
    end

    if containsAny(blob, self.TerminalWords) then
        score.TERMINAL += 55
    end

    if containsAny(blob, self.SwitchWords) then
        score.SWITCH += 55
    end

    if access.Locked then
        score.DOOR += 52
        strongDoor = true
    end

    if access.Type == "ITEM" then
        score.DOOR += 80
        strongDoor = true

    elseif access.Type == "CODE" then
        score.KEYPAD += 82
        score.DOOR += 35

    elseif access.Type == "SWITCH" then
        score.SWITCH += 75
        score.DOOR += 25

    elseif access.Type == "MULTI" then
        score.DOOR += 105
        score.KEYPAD += 35
        strongDoor = true
    end

    if prompt then
        local actionText = lower(prompt.ActionText)
        local objectText = lower(prompt.ObjectText)
        local combined = actionText .. " " .. objectText

        if containsAny(actionText, self.PickupWords)
            or containsAny(objectText, self.HighValueItemWords) then
            score.ITEM += 75
            strongItem = true
        end

        if containsAny(actionText, self.OpenWords)
            or containsAny(objectText, self.DoorWords) then
            score.DOOR += 48
            strongDoor = true
        end

        if containsAny(combined, self.CodeWords) then
            score.KEYPAD += 70
        end

        if containsAny(combined, self.SwitchWords) then
            score.SWITCH += 65
        end
    end

    if clickDetector then
        if score.ITEM > 0 then
            score.ITEM += 15
        end

        if score.DOOR > 0 then
            score.DOOR += 18
        end

        if score.KEYPAD > 0 then
            score.KEYPAD += 18
        end

        if score.TERMINAL > 0 then
            score.TERMINAL += 18
        end

        if score.SWITCH > 0 then
            score.SWITCH += 18
        end
    end

    if containsAny(blob, self.JunkWords)
        and not prompt
        and not access.Locked then
        score.DOOR -= 70
        score.SECRET_DOOR -= 70
    end

    local class = nil
    local best = -math.huge

    for candidateClass, candidateScore in pairs(score) do
        if candidateScore > best then
            class = candidateClass
            best = candidateScore
        end
    end

    if not class or best < State.Interactables.MinimumConfidence then
        return nil
    end

    if (
        class == "DOOR"
        or class == "SECRET_DOOR"
    )
        and State.Interactables.DoorMode ~= "All"
        and not strongDoor
        and best < 100 then
        return nil
    end

    return {
        Object = candidate,
        Class = class,
        Score = best,
        Access = access,
        Anchor = anchor,
        StrongItem = strongItem,
        StrongDoor = strongDoor,
    }
end

local Interactables = {
    Objects = {},
    RequirementIndex = {},
    Pinned = nil,

    Colors = {
        ITEM = Color3.fromRGB(255, 214, 92),
        DOOR = Color3.fromRGB(74, 200, 255),
        SECRET_DOOR = Color3.fromRGB(200, 100, 255),
        KEYPAD = Color3.fromRGB(97, 228, 210),
        TERMINAL = Color3.fromRGB(100, 168, 255),
        SWITCH = Color3.fromRGB(255, 167, 77),
    }

function Interactables:RebuildRequirementIndex()
    table.clear(self.RequirementIndex)

    local unique = {}

    for _, data in pairs(self.Objects) do
        local access = data.Info.Access

        if access and access.Item and access.Item ~= "" then
            local key = lower(access.Item)

            if not unique[key] then
                unique[key] = true
                table.insert(self.RequirementIndex, access.Item)
            end
        end
    end
end

function Interactables:GetRequirementMatch(data)
    if data.Info.Class ~= "ITEM" then
        return 0
    end

    local best = 0

    for _, requirement in ipairs(self.RequirementIndex) do
        best =
            math.max(
                best,
                nameMatchScore(
                    requirement,
                    data.Object.Name
                )
            )
    end

    return best
end

function Interactables:IsRelevantItem(data)
    if data.Info.Class ~= "ITEM" then
        return true
    end

    if State.Interactables.ItemMode == "All" then
        return true
    end

    local blob = Scanner:GetNameBlob(data.Object)
    local requirementMatch = self:GetRequirementMatch(data)

    if State.Interactables.ItemMode == "Useful" then
        return containsAny(blob, Scanner.HighValueItemWords)
            or data.Info.StrongItem
            or requirementMatch >= 50
    end

    return requirementMatch >= 45
end

function Interactables:DoorPassesMode(data)
    local class = data.Info.Class

    if class ~= "DOOR"
        and class ~= "SECRET_DOOR" then
        return true
    end

    if State.Interactables.DoorMode == "All" then
        return true
    end

    local access = data.Info.Access

    if State.Interactables.DoorMode == "Locked / Required" then
        return access.Locked
            or access.Type == "ITEM"
            or access.Type == "CODE"
            or access.Type == "SWITCH"
            or access.Type == "MULTI"
    end

    return data.Info.StrongDoor
        or access.Locked
        or access.Type ~= "FREE"
end

function Interactables:ShouldRenderClass(class)
    if class == "ITEM" then
        return State.Interactables.Items

    elseif class == "DOOR" then
        return State.Interactables.Doors

    elseif class == "SECRET_DOOR" then
        return State.Interactables.SecretDoors

    elseif class == "KEYPAD"
        or class == "TERMINAL"
        or class == "SWITCH" then
        return State.Interactables.AccessPoints
    end

    return false
end

function Interactables:Add(candidate)
    if not candidate then
        return false
    end

    if self.Objects[candidate] then
        return false
    end

    local info = Scanner:Classify(candidate)

    if not info then
        return false
    end

    self.Objects[candidate] = {
        Object = candidate,
        Info = info,
        Anchor = info.Anchor,
        Visual = nil,
        LastVisible = 0,
    }

    if info.Access and info.Access.Item then
        self:RebuildRequirementIndex()
    end

    return true
end

function Interactables:DestroyVisual(data)

end

function Interactables:ScanLocal()
    local root = Character:GetRoot()

    if not root then
        return 0
    end

    local before = self:GetCount()

    local params = OverlapParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {
        LocalPlayer.Character,
    }
    params.MaxParts = 2500

    local parts =
        Workspace:GetPartBoundsInRadius(
            root.Position,
            State.Interactables.LocalScanRadius,
            params
        )

    local seen = {}

    for _, part in ipairs(parts) do
        local candidate = Scanner:GetCandidate(part)

        if candidate and not seen[candidate] then
            seen[candidate] = true
            self:Add(candidate)
        end
    end

    for _, tag in ipairs({
        "Item", "Pickup", "Loot", "Key", "Collectible",
        "Door", "Gate", "VaultDoor",
        "SecretDoor", "HiddenDoor", "SecretEntrance",
        "Keypad", "CodePanel",
        "Terminal", "AccessPanel", "Console",
        "Switch", "Lever", "Generator", "Button",
    }) do
        for _, instance in ipairs(
            CollectionService:GetTagged(tag)
        ) do
            local candidate =
                Scanner:GetCandidate(instance)
                or instance

            self:Add(candidate)
        end
    end

    self:RebuildRequirementIndex()

    return self:GetCount() - before
end

function Interactables:FullScan()
    local before = self:GetCount()
    local descendants = Workspace:GetDescendants()

    for index, object in ipairs(descendants) do
        local candidate = nil

        if object:IsA("Tool")
            or object:IsA("ProximityPrompt")
            or object:IsA("ClickDetector") then

            candidate = Scanner:GetCandidate(object)

        elseif object:IsA("Model")
            or object:IsA("BasePart") then

            if Scanner:IsMeaningfulOwnName(object)
                or object:GetAttribute("Door") ~= nil
                or object:GetAttribute("Item") ~= nil
                or object:GetAttribute("Pickup") ~= nil
                or object:GetAttribute("Secret") ~= nil
                or object:GetAttribute("Hidden") ~= nil
                or object:GetAttribute("Keypad") ~= nil
                or object:GetAttribute("Terminal") ~= nil
                or object:GetAttribute("AccessPanel") ~= nil
                or object:GetAttribute("Switch") ~= nil
                or object:GetAttribute("RequiredItem") ~= nil
                or object:GetAttribute("RequiredCode") ~= nil
                or object:GetAttribute("AccessCode") ~= nil
                or object:GetAttribute("Password") ~= nil then

                candidate = Scanner:GetCandidate(object)
            end
        end

        if candidate then
            self:Add(candidate)
        end

        if index % 700 == 0 then
            RunService.Heartbeat:Wait()
        end
    end

    self:RebuildRequirementIndex()

    return self:GetCount() - before
end

function Interactables:FindByName(query)
    local root = Character:GetRoot()
    local best = nil
    local bestScore = -math.huge

    for object, data in pairs(self.Objects) do
        if object.Parent then
            local match =
                nameMatchScore(
                    query,
                    object.Name
                )

            if match >= 30 then
                local distance = math.huge

                if root
                    and data.Anchor
                    and data.Anchor.Parent then

                    distance =
                        (root.Position - data.Anchor.Position).Magnitude
                end

                local total =
                    match * 10
                    - math.min(distance, 5000) * 0.02

                if total > bestScore then
                    bestScore = total

                    best = {
                        Object = object,
                        Data = data,
                        Match = match,
                        Distance = distance,
                    }
                end
            end
        end
    end

    return best
end

function Interactables:SearchGlobal(query, requiredClass)
    local root = Character:GetRoot()
    local best = nil
    local bestScore = -math.huge
    local seen = {}

    local descendants = Workspace:GetDescendants()

    for index, object in ipairs(descendants) do
        local directMatch = 0
        local candidate = nil

        if object:IsA("BasePart")
            or object:IsA("Model")
            or object:IsA("Tool") then

            directMatch =
                nameMatchScore(
                    query,
                    object.Name
                )

            if directMatch >= 28 then
                candidate =
                    Scanner:GetCandidate(object)
                    or object
            end

        elseif object:IsA("ProximityPrompt") then
            directMatch =
                math.max(
                    nameMatchScore(
                        query,
                        object.ObjectText
                    ),
                    nameMatchScore(
                        query,
                        object.ActionText
                    )
                )

            if directMatch >= 28 then
                candidate = Scanner:GetCandidate(object)
            end
        end

        if candidate and not seen[candidate] then
            seen[candidate] = true

            local data = self.Objects[candidate]
            local info =
                data
                and data.Info
                or Scanner:Classify(candidate)

            if info
                and (
                    not requiredClass
                    or info.Class == requiredClass
                ) then

                if not data then
                    self:Add(candidate)
                    data = self.Objects[candidate]
                end

                if data then
                    local match =
                        math.max(
                            directMatch,
                            nameMatchScore(
                                query,
                                candidate.Name
                            )
                        )

                    local distance = math.huge

                    if root
                        and data.Anchor
                        and data.Anchor.Parent then

                        distance =
                            (root.Position - data.Anchor.Position).Magnitude
                    end

                    local total =
                        match * 10
                        + info.Score * 0.5
                        - math.min(distance, 5000) * 0.02

                    if total > bestScore then
                        bestScore = total

                        best = {
                            Object = candidate,
                            Data = data,
                            Match = match,
                            Distance = distance,
                        }
                    end
                end
            end
        end

        if index % 850 == 0 then
            RunService.Heartbeat:Wait()
        end
    end

    self:RebuildRequirementIndex()

    return best
end

function Interactables:FindBestItem(requirement)
    local localResult =
        self:FindByName(requirement)

    if localResult
        and localResult.Data.Info.Class == "ITEM"
        and localResult.Match >= 40 then
        return localResult
    end

    return self:SearchGlobal(
        requirement,
        "ITEM"
    )
end

function Interactables:GetNearestRequiredAccess()
    local root = Character:GetRoot()

    if not root then
        return nil
    end

    local best = nil
    local bestDistance = math.huge

    for object, data in pairs(self.Objects) do
        local class = data.Info.Class
        local access = data.Info.Access

        if (
            class == "DOOR"
            or class == "SECRET_DOOR"
            or class == "KEYPAD"
            or class == "TERMINAL"
            or class == "SWITCH"
        )
            and access
            and (
                access.Locked
                or access.Type ~= "FREE"
            )
            and data.Anchor
            and data.Anchor.Parent then

            local distance =
                (root.Position - data.Anchor.Position).Magnitude

            if distance < bestDistance then
                bestDistance = distance

                best = {
                    Object = object,
                    Data = data,
                    Distance = distance,
                }
            end
        end
    end

    return best
end

local function directionArrow(worldPosition)
    Camera = Workspace.CurrentCamera

    if not Camera then
        return "?"
    end

    local localPosition =
        Camera.CFrame:PointToObjectSpace(worldPosition)

    local angle =
        math.atan2(
            localPosition.X,
            -localPosition.Z
        )

    local directions = {
        "N", "NE", "E", "SE",
        "S", "SW", "W", "NW",
    }

    local index =
        (math.floor(
            (angle + math.pi / 8)
            / (math.pi / 4)
        ) % 8) + 1

    return directions[index]
end

local Inspector = {
    Enabled = false,
    Last = nil,
    DialogCounter = 0,

function Inspector:GetTargetAt(screenPosition)
    Camera = Workspace.CurrentCamera

    if not Camera then
        return nil
    end

    local ray =
        Camera:
        ViewportPointToRay(
            screenPosition.X,
            screenPosition.Y
        )

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {
        LocalPlayer.Character,
    }

    local result =
        Workspace:Raycast(
            ray.Origin,
            ray.Direction * 1500,
            params
        )

    return result and result.Instance or nil
end

function Inspector:GetCenterTarget()
    Camera = Workspace.CurrentCamera

    if not Camera then
        return nil
    end

    local size = Camera.ViewportSize

    return self:GetTargetAt(
        Vector2.new(
            size.X / 2,
            size.Y / 2
        )
    )
end

function Inspector:Resolve(target)
    local current = target

    for _ = 1, 7 do
        if not current or current == Workspace then
            break
        end

        local candidate =
            Scanner:GetCandidate(current)
            or current

        if candidate
            and candidate:IsDescendantOf(Workspace) then

            local known = Interactables.Objects[candidate]

            if known then
                return candidate, known.Info
            end

            local info = Scanner:Classify(candidate)

            if info then
                Interactables:Add(candidate)
                return candidate, info
            end
        end

        current = current.Parent
    end

    return nil
end

function Inspector:BuildDescription(candidate, info)
    local root = Character:GetRoot()
    local anchor =
        info.Anchor
        or Scanner:GetAnchor(candidate)

    local distance = nil

    if root and anchor then
        distance =
            (root.Position - anchor.Position).Magnitude
    end

    local lines = {
        string.format(
            "%s | confidence %d",
            prettyClass(info.Class),
            round(info.Score)
        )
    }

    if distance then
        table.insert(
            lines,
            string.format(
                "%d studs away",
                round(distance)
            )
        )
    end

    local access = info.Access

    if access then
        if access.Locked then
            table.insert(
                lines,
                "Status: Locked"
            )
        end

        if access.Type ~= "FREE"
            and access.Type ~= "UNKNOWN" then

            table.insert(
                lines,
                "Access: " .. access.Type
            )
        end

        if access.Item then
            table.insert(
                lines,
                "Needs: " .. access.Item
            )

            local match =
                Interactables:
                FindByName(
                    access.Item
                )

            if match
                and match.Data.Info.Class == "ITEM" then

                table.insert(
                    lines,
                    string.format(
                        "Indexed match: %s | %d studs",
                        match.Object.Name,
                        round(match.Distance)
                    )
                )

                if State.Interactables.AutoPinRequiredItem then
                    Interactables:Pin(match.Object)
                end
            end
        end

        if access.Code then
            table.insert(
                lines,
                "Code: " .. access.Code
            )

        elseif access.Type == "CODE"
            or access.Type == "MULTI" then

            table.insert(
                lines,
                "Code: Unknown / Server-side"
            )
        end

        if access.Trigger then
            table.insert(
                lines,
                "Trigger: " .. access.Trigger
            )
        end
    end

    local tags = CollectionService:GetTags(candidate)

    if #tags > 0 then
        table.insert(
            lines,
            "Tags: "
            .. table.concat(tags, ", ")
        )
    end

    return table.concat(lines, "\n")
end

local Frames = 0
local LastFPS = os.clock()

connect(RunService.RenderStepped, function()
    Frames += 1

    local now = os.clock()

    if now - LastFPS >= 1 then
        State.Performance.FPS =
            math.floor(
                Frames
                / (
                    now
                    - LastFPS
                )
            )

        Frames = 0
        LastFPS = now

        if State.Performance.Adaptive then
            if State.Performance.FPS < 32 then
                State.Interactables.UpdateRate = 0.30
                State.EntityESP.UpdateRate = 0.30

            elseif State.Performance.FPS < 42 then
                State.Interactables.UpdateRate = 0.22
                State.EntityESP.UpdateRate = 0.22

            elseif State.Performance.FPS > 52 then
                State.Interactables.UpdateRate = 0.16
                State.EntityESP.UpdateRate = 0.18
            end
        end
    end
end)

local LastEntityRender = 0
local LastInteractableRender = 0
local LastCleanup = 0
local LastHUD = 0

connect(RunService.RenderStepped, function()
    if not State.Running then
        return
    end

    local now = os.clock()

    -- Distinct entity ESP budgets prevent monsters from hiding NPC labels.
    if now - LastEntityRender >= State.EntityESP.UpdateRate then
        LastEntityRender = now

        local playerRoot = Character:GetRoot()

        if playerRoot then
            local buckets = {
                PLAYER = {},
                NPC = {},
                MONSTER = {},
                ENTITY = {},
            }

            local remove = {}
            local nearestMonster = nil

            for model, data in pairs(EntityESP.Objects) do
                if not model.Parent then
                    table.insert(remove, model)
                    continue
                end

                local root = data.Root

                if not root or not root.Parent then
                    root = getEntityRoot(model)
                    data.Root = root

                    if data.Visual and root then
                        data.Visual.Gui.Adornee = root
                    end
                end

                if not root then
                    EntityTracker:Queue(model)
                    continue
                end

                -- Periodically reclassify. This catches rigs that gain tags,
                -- attributes, Humanoids, or hostility after initial spawn.
                if now - (data.LastClassCheck or 0) >= 2.0
                    and data.Class ~= "PLAYER" then
                    EntityESP:AddOrUpdate(model)
                    data = EntityESP.Objects[model] or data
                end

                if data.Visual then
                    data.Visual.Highlight.Enabled = false
                    data.Visual.Gui.Enabled = false
                end

                if EntityESP:ShouldShow(data.Class) then
                    local distance =
                        (playerRoot.Position - root.Position).Magnitude

                    if distance <= State.EntityESP.MaxDistance then
                        local bucket = buckets[data.Class]

                        if bucket then
                            table.insert(bucket, {
                                Data = data,
                                Distance = distance,
                            })
                        end

                        if data.Class == "MONSTER"
                            and (
                                not nearestMonster
                                or distance < nearestMonster.Distance
                            ) then
                            nearestMonster = {
                                Data = data,
                                Distance = distance,
                            }
                        end
                    end
                end
            end

            for class, bucket in pairs(buckets) do
                table.sort(bucket, function(a, b)
                    return a.Distance < b.Distance
                end)

                local budget = EntityESP:GetBudget(class)

                for index = 1, math.min(#bucket, budget) do
                    local candidate = bucket[index]
                    local data = candidate.Data
                    local visual = EntityESP:EnsureVisual(data)

                    if visual then
                        data.LastVisible = now
                        visual.Highlight.Enabled = true
                        visual.Gui.Enabled = true
                        EntityESP:UpdateVisual(data, candidate.Distance)
                    end
                end
            end

            if State.EntityESP.ThreatHUD then
                if nearestMonster then
                    local humanoid = getEntityHumanoid(nearestMonster.Data.Model)
                    local hpText = ""

                    if humanoid then
                        local ratio = clamp(
                            humanoid.Health / math.max(humanoid.MaxHealth, 1),
                            0,
                            1
                        )

                        hpText = " | " .. tostring(round(ratio * 100)) .. "% HP"
                    end

                    ThreatHUD:SetText(
                        "THREAT | "
                        .. nearestMonster.Data.Model.Name
                        .. " | "
                        .. tostring(round(nearestMonster.Distance))
                        .. "st"
                        .. hpText
                    )
                    ThreatHUD:SetVisible(true)
                else
                    ThreatHUD:SetText("THREAT | clear")
                    ThreatHUD:SetVisible(true)
                end
            else
                ThreatHUD:SetVisible(false)
            end

            for _, model in ipairs(remove) do
                EntityESP:Remove(model)
            end
        end
    end

    -- Interactable ESP uses a priority queue and a strict visual budget.
    if now - LastInteractableRender >= State.Interactables.UpdateRate then
        LastInteractableRender = now

        local root = Character:GetRoot()

        if root then
            local candidates = {}
            local remove = {}

            for object, data in pairs(Interactables.Objects) do
                if not object.Parent then
                    table.insert(remove, object)
                    continue
                end

                local anchor = data.Anchor

                if not anchor or not anchor.Parent then
                    anchor = Scanner:GetAnchor(object)
                    data.Anchor = anchor

                    if data.Visual and anchor then
                        data.Visual.Gui.Adornee = anchor
                    end
                end

                if not anchor then
                    continue
                end

                if data.Visual then
                    data.Visual.Highlight.Enabled = false
                    data.Visual.Gui.Enabled = false
                end

                local pinned = Interactables.Pinned == object

                local classEnabled =
                    Interactables:ShouldRenderClass(data.Info.Class)
                    or pinned

                local itemPass =
                    Interactables:IsRelevantItem(data)
                    or pinned

                local doorPass =
                    Interactables:DoorPassesMode(data)
                    or pinned

                if classEnabled and itemPass and doorPass then
                    local distance =
                        (root.Position - anchor.Position).Magnitude

                    if distance <= State.Interactables.MaxDistance
                        or (
                            pinned
                            and State.Interactables.PinnedAlwaysVisible
                        ) then

                        local priority = 0

                        if pinned then
                            priority += 100000
                        end

                        if data.Info.Class == "ITEM" then
                            priority += 2500
                        elseif data.Info.Class == "KEYPAD" then
                            priority += 1850
                        elseif data.Info.Class == "SECRET_DOOR" then
                            priority += 1750
                        elseif data.Info.Class == "TERMINAL" then
                            priority += 1550
                        elseif data.Info.Class == "SWITCH" then
                            priority += 1450
                        elseif data.Info.Class == "DOOR" then
                            priority += 900
                        end

                        local access = data.Info.Access

                        if access and access.Item then
                            priority += 1900
                        end

                        if access and access.Locked then
                            priority += 850
                        end

                        if data.Info.Class == "ITEM" then
                            priority +=
                                Interactables:GetRequirementMatch(data)
                                * 20
                        end

                        priority += data.Info.Score * 4
                        priority -= math.min(distance, 5000)

                        table.insert(candidates, {
                            Data = data,
                            Distance = distance,
                            Priority = priority,
                            Pinned = pinned,
                        })
                    end
                end
            end

            table.sort(candidates, function(a, b)
                if a.Pinned ~= b.Pinned then
                    return a.Pinned
                end

                return a.Priority > b.Priority
            end)

            for index = 1,
                math.min(
                    #candidates,
                    State.Interactables.MaxVisible
                ) do

                local candidate = candidates[index]
                local data = candidate.Data
                local visual = Interactables:EnsureVisual(data)

                if visual then
                    data.LastVisible = now
                    visual.Highlight.Enabled = true
                    visual.Gui.Enabled = true

                    local text = data.Object.Name
                    local access = data.Info.Access

                    local detailed =
                        candidate.Pinned
                        or candidate.Distance
                            <= State.Interactables.DetailDistance

                    if detailed
                        and State.Interactables.ShowRequirement
                        and access then

                        if access.Item then
                            text ..= "\nKEY: " .. access.Item
                        elseif access.Type == "CODE" then
                            text ..= "\nCODE REQUIRED"
                        elseif access.Type == "SWITCH" then
                            text ..= "\nTRIGGER REQUIRED"
                        elseif access.Type == "MULTI" then
                            text ..= "\nMULTI ACCESS"
                        end
                    end

                    if State.Interactables.ShowDistance then
                        text ..=
                            " | "
                            .. tostring(round(candidate.Distance))
                            .. "st"
                    end

                    if State.Interactables.ShowConfidence then
                        text ..=
                            " ["
                            .. tostring(round(data.Info.Score))
                            .. "]"
                    end

                    if candidate.Pinned then
                        text = "[PIN] " .. text
                        visual.Highlight.OutlineTransparency = 0
                    else
                        visual.Highlight.OutlineTransparency = 0.08
                    end

                    visual.Label.Text = text
                end
            end

            for _, object in ipairs(remove) do
                Interactables:Remove(object)
            end
        end
    end

    if now - LastHUD >= 0.20 then
        LastHUD = now

        if State.Performance.ShowHUD then
            PerformanceHUD:SetText(
                string.format(
                    "Nuvem | %d FPS | %d objects",
                    State.Performance.FPS,
                    EntityESP:GetCount() + Interactables:GetCount()
                )
            )
        end

        local pinned = Interactables.Pinned

        if pinned
            and pinned.Parent
            and Interactables.Objects[pinned] then

            local data = Interactables.Objects[pinned]
            local anchor = data.Anchor
            local root = Character:GetRoot()

            if anchor and root then
                TargetHUD:SetText(
                    string.format(
                        "%s | %s | %dst",
                        directionArrow(anchor.Position),
                        pinned.Name,
                        round(
                            (root.Position - anchor.Position).Magnitude
                        )
                    )
                )

                TargetHUD:SetVisible(true)
            end
        else
            TargetHUD:SetVisible(false)
        end
    end

    if now - LastCleanup >= 5 then
        LastCleanup = now

        for _, data in pairs(Interactables.Objects) do
            if data.Visual
                and now - data.LastVisible > 12 then
                Interactables:DestroyVisual(data)
            end
        end

        for _, data in pairs(EntityESP.Objects) do
            if data.Visual
                and now - data.LastVisible > 12 then
                EntityESP:DestroyVisual(data)
            end
        end
    end
end)

connect(Workspace.DescendantAdded, function(object)
    if not State.Running then
        return
    end

    if object:IsA("Model")
        or object:IsA("Humanoid")
        or object:IsA("AnimationController") then
        EntityTracker:Queue(object)
    end

    if object:IsA("Tool")
        or object:IsA("ProximityPrompt")
        or object:IsA("ClickDetector") then

        task.delay(0.12, function()
            if not State.Running then
                return
            end

            local candidate = Scanner:GetCandidate(object)

            if candidate then
                Interactables:Add(candidate)
                Interactables:RebuildRequirementIndex()
            end
        end)
    end
end)

-- This small spatial sweep is the main reliability fix. It catches rigs that
-- were inserted incomplete, received their Humanoid later, or changed tags
-- after the original spawn event. It only checks the area that could be drawn.
task.spawn(function()
    while State.Running do
        task.wait(State.EntityESP.RecoveryInterval)

        if not State.Running then
            break
        end

        if State.EntityESP.AutoRecovery
            and (
                State.EntityESP.Players
                or State.EntityESP.NPCs
                or State.EntityESP.Monsters
                or State.EntityESP.Entities
                or State.EntityESP.ThreatHUD
            ) then

            pcall(function()
                EntityTracker:ScanNearby()
            end)
        end
    end
end)

local function hookPlayerEntity(player)
    connect(player.CharacterAdded, function(character)
        task.wait(0.15)

        if player == LocalPlayer then
            restoreCollision()
            Character:CaptureOriginal()
            Character:Apply()
        else
            local root = getEntityRoot(character)

            if root then
                EntityESP.Objects[character] = {
                    Model = character,
                    Class = "PLAYER",
                    Root = root,
                    Visual = nil,
                    LastVisible = 0,
                    LastClassCheck = os.clock(),
                }
            end
        end
    end)

    if player ~= LocalPlayer and player.Character then
        local root = getEntityRoot(player.Character)

        if root then
            EntityESP.Objects[player.Character] = {
                Model = player.Character,
                Class = "PLAYER",
                Root = root,
                Visual = nil,
                LastVisible = 0,
                LastClassCheck = os.clock(),
            }
        end
    end
end

    TweenService:Create(
        Lighting,
        TweenInfo.new(
            0.28,
            Enum.EasingStyle.Quart,
            Enum.EasingDirection.Out
        ),
        {
            Brightness =
                State.World.Fullbright
                and 2.5
                or OriginalLighting.Brightness,

            ClockTime =
                State.World.TimeOverride
                and State.World.Time
                or OriginalLighting.ClockTime,

            FogEnd =
                State.World.NoFog
                and 100000
                or OriginalLighting.FogEnd,
        }
    ):Play()

    if State.World.Fullbright then
        Lighting.Ambient =
            Color3.fromRGB(
                180,
                180,
                180
            )

        Lighting.OutdoorAmbient =
            Color3.fromRGB(
                170,
                170,
                170
            )

        Lighting.GlobalShadows = false
    else
        Lighting.Ambient = OriginalLighting.Ambient
        Lighting.OutdoorAmbient = OriginalLighting.OutdoorAmbient
        Lighting.GlobalShadows = OriginalLighting.GlobalShadows
    end
end

    Workspace.Gravity =
        State.World.GravityOverride
        and State.World.Gravity
        or OriginalWorld.Gravity
end

                Enum.EasingDirection.Out

    end

local function getMouseRay(screenPosition, distance)
    Camera = Workspace.CurrentCamera

    if not Camera then
        return nil
    end

    local ray = Camera:ViewportPointToRay(screenPosition.X, screenPosition.Y)
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = { LocalPlayer.Character }

    return Workspace:Raycast(
        ray.Origin,
        ray.Direction * (distance or 5000),
        params
    )
end

local function findTargetModel(query)
    query = lower(query)

    if query == "" then
        return nil
    end

    local best = nil
    local bestScore = -math.huge
    local playerRoot = Character:GetRoot()

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local root = getEntityRoot(player.Character)

            if root then
                local match = math.max(
                    nameMatchScore(query, player.Name),
                    nameMatchScore(query, player.DisplayName)
                )

                if match >= 25 then
                    local distance = playerRoot
                        and (playerRoot.Position - root.Position).Magnitude
                        or 0

                    local score = match * 10 - distance * 0.01

                    if score > bestScore then
                        bestScore = score
                        best = player.Character
                    end
                end
            end
        end
    end

    for model, data in pairs(EntityESP.Objects) do
        if model.Parent and data.Root and data.Root.Parent then
            local match = nameMatchScore(query, model.Name)

            if match >= 25 then
                local distance = playerRoot
                    and (playerRoot.Position - data.Root.Position).Magnitude
                    or 0

                local score = match * 10 - distance * 0.01

                if score > bestScore then
                    bestScore = score
                    best = model
                end
            end
        end
    end

    return best
end

local WaypointManager = {
    Points = {},
}

function WaypointManager:Save(name)
    local root = Character:GetRoot()

    if not root then
        return false, "Character root is unavailable."
    end

    name = tostring(name or "Waypoint")

    if name == "" then
        name = "Waypoint"
    end

    self.Points[name] = root.CFrame
    return true, name
end

function WaypointManager:Go(name)
    local root = Character:GetRoot()
    local cframe = self.Points[tostring(name or "")]

    if not root then
        return false, "Character root is unavailable."
    end

    if not cframe then
        return false, "Waypoint not found."
    end

    root.CFrame = cframe
    return true
end

function WaypointManager:Delete(name)
    name = tostring(name or "")

    if self.Points[name] then
        self.Points[name] = nil
        return true
    end

    return false
end

function WaypointManager:List()
    local names = {}

    for name in pairs(self.Points) do
        table.insert(names, name)
    end

    table.sort(names)
    return names
end

function WaypointManager:SaveFile()
    if not HasFS then
        return false, "Filesystem APIs are unavailable."
    end

    ensureFolder()

    local encoded = {}

    for name, cframe in pairs(self.Points) do
        encoded[name] = { cframe:GetComponents() }
    end

    writefile(
        CACHE_FOLDER .. "/Waypoints_" .. tostring(game.PlaceId) .. ".json",
        HttpService:JSONEncode(encoded)
    )

    return true
end

function WaypointManager:LoadFile()
    if not HasFS then
        return false, "Filesystem APIs are unavailable."
    end

    local path = CACHE_FOLDER .. "/Waypoints_" .. tostring(game.PlaceId) .. ".json"

    if not isfile(path) then
        return false, "No saved waypoint file."
    end

    local ok, decoded = pcall(function()
        return HttpService:JSONDecode(readfile(path))
    end)

    if not ok or type(decoded) ~= "table" then
        return false, "Waypoint file is invalid."
    end

    for name, components in pairs(decoded) do
        if type(components) == "table" and #components >= 12 then
            self.Points[name] = CFrame.new(table.unpack(components))
        end
    end

    return true
end

local Flight = {
    Velocity = nil,
    Gyro = nil,
}

function Flight:Enable()
    if State.Mobility.Flight then
        return
    end

    local root = Character:GetRoot()
    local humanoid = Character:GetHumanoid()

    if not root or not humanoid then
        card("WARNING", "Flight", "Character is not ready.", 4)
        return
    end

    State.Mobility.Flight = true

    local velocity = Instance.new("BodyVelocity")
    velocity.Name = "_NuvemFlightVelocity"
    velocity.MaxForce = Vector3.new(1e9, 1e9, 1e9)
    velocity.P = 25000
    velocity.Velocity = Vector3.zero
    velocity.Parent = root

    local gyro = Instance.new("BodyGyro")
    gyro.Name = "_NuvemFlightGyro"
    gyro.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
    gyro.P = 35000
    gyro.D = 900
    gyro.CFrame = root.CFrame
    gyro.Parent = root

    self.Velocity = velocity
    self.Gyro = gyro

    humanoid.AutoRotate = false
end

function Flight:Disable()
    State.Mobility.Flight = false

    if self.Velocity then
        self.Velocity:Destroy()
        self.Velocity = nil
    end

    if self.Gyro then
        self.Gyro:Destroy()
        self.Gyro = nil
    end

    local humanoid = Character:GetHumanoid()

    if humanoid then
        humanoid.AutoRotate = true
    end
end

function Flight:Set(value)
    if value then
        self:Enable()
    else
        self:Disable()
    end
end

connect(RunService.RenderStepped, function()
    if not State.Running
        or not State.Mobility.Flight
        or not Flight.Velocity
        or not Flight.Gyro then
        return
    end

    local root = Character:GetRoot()
    local humanoid = Character:GetHumanoid()
    Camera = Workspace.CurrentCamera

    if not root or not humanoid or not Camera then
        return
    end

    local move = humanoid.MoveDirection
    local cameraCF = Camera.CFrame
    local horizontal = Vector3.zero

    if move.Magnitude > 0 then
        local look = Vector3.new(cameraCF.LookVector.X, 0, cameraCF.LookVector.Z)
        local right = Vector3.new(cameraCF.RightVector.X, 0, cameraCF.RightVector.Z)

        if look.Magnitude > 0 then
            look = look.Unit
        end

        if right.Magnitude > 0 then
            right = right.Unit
        end

        local objectMove = cameraCF:VectorToObjectSpace(move)
        horizontal = right * objectMove.X - look * objectMove.Z
    end

    local vertical = 0

    if UIS:IsKeyDown(Enum.KeyCode.Space) or humanoid.Jump then
        vertical += State.Mobility.FlightVerticalSpeed
    end

    if UIS:IsKeyDown(Enum.KeyCode.LeftControl)
        or UIS:IsKeyDown(Enum.KeyCode.C) then
        vertical -= State.Mobility.FlightVerticalSpeed
    end

    Flight.Velocity.Velocity =
        horizontal * State.Mobility.FlightSpeed
        + Vector3.new(0, vertical, 0)

    local flatLook = Vector3.new(
        cameraCF.LookVector.X,
        0,
        cameraCF.LookVector.Z
    )

    if flatLook.Magnitude > 0.01 then
        Flight.Gyro.CFrame = CFrame.lookAt(
            root.Position,
            root.Position + flatLook.Unit
        )
    end
end)

local Freecam = {
    Saved = nil,
    Position = nil,
    Yaw = 0,
    Pitch = 0,
}

function Freecam:Enable()
    if State.Freecam.Enabled then
        return
    end

    Camera = Workspace.CurrentCamera

    if not Camera then
        return
    end

    State.Freecam.Enabled = true

    self.Saved = {
        CameraType = Camera.CameraType,
        CameraSubject = Camera.CameraSubject,
        CFrame = Camera.CFrame,
        Focus = Camera.Focus,
        MouseBehavior = UIS.MouseBehavior,
        MouseIconEnabled = UIS.MouseIconEnabled,
    }

    self.Position = Camera.CFrame.Position

    local look = Camera.CFrame.LookVector
    self.Yaw = math.atan2(-look.X, -look.Z)
    self.Pitch = math.asin(clamp(look.Y, -1, 1))

    Camera.CameraType = Enum.CameraType.Scriptable

    if not Library.IsMobile then
        UIS.MouseBehavior = Enum.MouseBehavior.LockCenter
        UIS.MouseIconEnabled = false
    end
end

function Freecam:Disable()
    if not State.Freecam.Enabled then
        return
    end

    State.Freecam.Enabled = false
    Camera = Workspace.CurrentCamera

    if Camera and self.Saved then
        Camera.CameraType = self.Saved.CameraType
        Camera.CameraSubject = self.Saved.CameraSubject
        Camera.CFrame = self.Saved.CFrame
        Camera.Focus = self.Saved.Focus
    end

    if self.Saved then
        UIS.MouseBehavior = self.Saved.MouseBehavior
        UIS.MouseIconEnabled = self.Saved.MouseIconEnabled
    end

    self.Saved = nil
end

function Freecam:Set(value)
    if value then
        self:Enable()
    else
        self:Disable()
    end
end

connect(RunService.RenderStepped, function(dt)
    if not State.Running or not State.Freecam.Enabled then
        return
    end

    Camera = Workspace.CurrentCamera

    if not Camera then
        return
    end

    if not Library.IsMobile then
        local delta = UIS:GetMouseDelta()
        Freecam.Yaw -= math.rad(delta.X * State.Freecam.Sensitivity)
        Freecam.Pitch = clamp(
            Freecam.Pitch - math.rad(delta.Y * State.Freecam.Sensitivity),
            math.rad(-89),
            math.rad(89)
        )
    end

    local rotation =
        CFrame.Angles(0, Freecam.Yaw, 0)
        * CFrame.Angles(Freecam.Pitch, 0, 0)

    local speed = State.Freecam.Speed

    if UIS:IsKeyDown(Enum.KeyCode.LeftShift) then
        speed *= State.Freecam.Boost
    end

    local move = Vector3.zero

    if UIS:IsKeyDown(Enum.KeyCode.W) then move += Vector3.new(0, 0, -1) end
    if UIS:IsKeyDown(Enum.KeyCode.S) then move += Vector3.new(0, 0, 1) end
    if UIS:IsKeyDown(Enum.KeyCode.A) then move += Vector3.new(-1, 0, 0) end
    if UIS:IsKeyDown(Enum.KeyCode.D) then move += Vector3.new(1, 0, 0) end
    if UIS:IsKeyDown(Enum.KeyCode.E) then move += Vector3.new(0, 1, 0) end
    if UIS:IsKeyDown(Enum.KeyCode.Q) then move += Vector3.new(0, -1, 0) end

    if move.Magnitude > 0 then
        move = move.Unit
    end

    Freecam.Position += rotation:VectorToWorldSpace(move) * speed * dt

    Camera.CFrame = CFrame.new(Freecam.Position) * rotation
    Camera.Focus = Camera.CFrame * CFrame.new(0, 0, -12)
end)

local XRay = {
    Originals = setmetatable({}, { __mode = "k" }),
    LastScan = 0,
}

function XRay:Restore()
    for part, value in pairs(self.Originals) do
        if part and part.Parent then
            part.LocalTransparencyModifier = value
        end
    end

    table.clear(self.Originals)
end

function XRay:Update()
    if not State.Utility.XRay then
        self:Restore()
        return
    end

    local root = Character:GetRoot()

    if not root then
        return
    end

    local params = OverlapParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = { LocalPlayer.Character }
    params.MaxParts = 1800

    local parts = Workspace:GetPartBoundsInRadius(
        root.Position,
        State.Utility.XRayRadius,
        params
    )

    for _, part in ipairs(parts) do
        if part:IsA("BasePart")
            and part.Anchored
            and part.CanCollide
            and part.Transparency < 0.95
            and not part:FindFirstAncestorWhichIsA("Tool") then

            if self.Originals[part] == nil then
                self.Originals[part] = part.LocalTransparencyModifier
            end

            part.LocalTransparencyModifier = math.max(
                self.Originals[part],
                State.Utility.XRayTransparency
            )
        end
    end
end

connect(RunService.Heartbeat, function()
    if not State.Running then
        return
    end

    local now = os.clock()

    if State.Utility.XRay and now - XRay.LastScan >= 0.8 then
        XRay.LastScan = now
        XRay:Update()
    end
end)

local RadarHUD = Library:AddDraggableLabel({
    Text = "Radar offline",
    Icon = "radar",
    IconPosition = "Left",
})

RadarHUD:SetVisible(false)

local function compassDirection(worldPosition)
    Camera = Workspace.CurrentCamera

    if not Camera then
        return "?"
    end

    local localPos = Camera.CFrame:PointToObjectSpace(worldPosition)
    local angle = math.deg(math.atan2(localPos.X, -localPos.Z))

    if angle >= -22.5 and angle < 22.5 then return "N" end
    if angle >= 22.5 and angle < 67.5 then return "NE" end
    if angle >= 67.5 and angle < 112.5 then return "E" end
    if angle >= 112.5 and angle < 157.5 then return "SE" end
    if angle >= 157.5 or angle < -157.5 then return "S" end
    if angle >= -157.5 and angle < -112.5 then return "SW" end
    if angle >= -112.5 and angle < -67.5 then return "W" end
    return "NW"
end

local LastRadarUpdate = 0

connect(RunService.RenderStepped, function()
    if not State.Running or not State.Utility.Radar then
        return
    end

    local now = os.clock()

    if now - LastRadarUpdate < 0.25 then
        return
    end

    LastRadarUpdate = now

    local root = Character:GetRoot()

    if not root then
        return
    end

    local entries = {}

    for model, data in pairs(EntityESP.Objects) do
        if model.Parent and data.Root and data.Root.Parent then
            local distance = (root.Position - data.Root.Position).Magnitude

            if distance <= State.Utility.RadarRange
                and (
                    data.Class == "PLAYER"
                    or data.Class == "MONSTER"
                    or data.Class == "NPC"
                ) then

                table.insert(entries, {
                    Model = model,
                    Data = data,
                    Distance = distance,
                })
            end
        end
    end

    table.sort(entries, function(a, b)
        return a.Distance < b.Distance
    end)

    local lines = { "RADAR" }

    for index = 1, math.min(#entries, State.Utility.RadarCount) do
        local entry = entries[index]
        local name = entry.Model.Name

        if State.Privacy.StreamerMode and entry.Data.Class == "PLAYER" then
            name = "PLAYER"
        end

        table.insert(
            lines,
            string.format(
                "%s %s | %s | %dst",
                compassDirection(entry.Data.Root.Position),
                prettyClass(entry.Data.Class),
                name,
                round(entry.Distance)
            )
        )
    end

    if #entries == 0 then
        table.insert(lines, "No tracked entities nearby")
    end

    RadarHUD:SetText(table.concat(lines, "\n"))
end)

    if not State.Running then

    end

        and not State.Utility.MonsterTracers then
        Tracers:Clear()
        return
    end

    end

    LastTracerUpdate = now
    Camera = Workspace.CurrentCamera

    if not Camera then
        return
    end

    local screen = Camera.ViewportSize
    local origin = Vector2.new(screen.X / 2, screen.Y - 22)
    local candidates = {}

    for model, data in pairs(EntityESP.Objects) do
        local allowed =
            (data.Class == "PLAYER" and State.Utility.PlayerTracers)
            or (data.Class == "MONSTER" and State.Utility.MonsterTracers)

        if allowed and model.Parent and data.Root and data.Root.Parent then
            local point, visible = Camera:WorldToViewportPoint(data.Root.Position)

            if visible and point.Z > 0 then
                table.insert(candidates, {
                    Position = Vector2.new(point.X, point.Y),
                    Class = data.Class,
                    Distance = point.Z,
                })
            end
        end
    end

    table.sort(candidates, function(a, b)
        return a.Distance < b.Distance
    end)

    for index = 1, math.max(#Tracers.Pool, maxLines) do

        local entry = candidates[index]

        if entry then
            local delta = entry.Position - origin
            local length = delta.Magnitude

        else
            line.Visible = false
        end
    end

local Macro = {}

function Macro:Clear()
    State.Macro.Recording = false
    State.Macro.Playing = false
    table.clear(State.Macro.Events)
end

function Macro:StartRecording()
    if State.Macro.Playing then
        return false, "Stop playback first."
    end

    table.clear(State.Macro.Events)
    State.Macro.Recording = true
    State.Macro.StartedAt = os.clock()

    return true
end

function Macro:StopRecording()
    State.Macro.Recording = false
end

function Macro:AddEvent(event)
    if not State.Macro.Recording
        or #State.Macro.Events >= State.Macro.MaxEvents then
        return
    end

    event.Time = os.clock() - State.Macro.StartedAt
    table.insert(State.Macro.Events, event)
end

function Macro:AddTeleportSnapshot()
    local root = Character:GetRoot()

    if not root then
        return false
    end

    local event = {
        Type = "CFrame",
        Value = { root.CFrame:GetComponents() },
        Time = State.Macro.Recording
            and (os.clock() - State.Macro.StartedAt)
            or (#State.Macro.Events > 0 and State.Macro.Events[#State.Macro.Events].Time or 0),
    }

    table.insert(State.Macro.Events, event)
    return true
end

local function macroSend(event)
    if event.Type == "Key" and VirtualInputManager then
        local key = Enum.KeyCode[event.Key]

        if key then
            pcall(function()
                VirtualInputManager:SendKeyEvent(
                    event.Down,
                    key,
                    false,
                    game
                )
            end)
        end

    elseif event.Type == "Mouse1" and VirtualInputManager then
        pcall(function()
            VirtualInputManager:SendMouseButtonEvent(
                event.X or 0,
                event.Y or 0,
                0,
                event.Down,
                game,
                0
            )
        end)

    elseif event.Type == "CFrame" then
        local root = Character:GetRoot()

        if root and type(event.Value) == "table" and #event.Value >= 12 then
            root.CFrame = CFrame.new(table.unpack(event.Value))
        end
    end
end

function Macro:Play()
    if State.Macro.Recording or State.Macro.Playing then
        return false
    end

    if #State.Macro.Events == 0 then
        return false
    end

    State.Macro.Playing = true

    task.spawn(function()
        repeat
            local previous = 0

            for _, event in ipairs(State.Macro.Events) do
                if not State.Running or not State.Macro.Playing then
                    break
                end

                local waitTime = math.max(
                    0,
                    (event.Time - previous) / math.max(State.Macro.Speed, 0.05)
                )

                if waitTime > 0 then
                    task.wait(waitTime)
                end

                macroSend(event)
                previous = event.Time
            end
        until not State.Macro.Loop
            or not State.Macro.Playing
            or not State.Running

        State.Macro.Playing = false
    end)

    return true
end

function Macro:Stop()
    State.Macro.Playing = false
end

function Macro:Save(name)
    if not HasFS then
        return false, "Filesystem APIs are unavailable."
    end

    ensureFolder()

    name = tostring(name or "macro")
    name = string.gsub(name, "[^%w_%-]", "_")

    local path = CACHE_FOLDER .. "/Macro_" .. name .. ".json"
    writefile(path, HttpService:JSONEncode(State.Macro.Events))

    return true, path
end

function Macro:Load(name)
    if not HasFS then
        return false, "Filesystem APIs are unavailable."
    end

    name = tostring(name or "macro")
    name = string.gsub(name, "[^%w_%-]", "_")

    local path = CACHE_FOLDER .. "/Macro_" .. name .. ".json"

    if not isfile(path) then
        return false, "Macro file not found."
    end

    local ok, decoded = pcall(function()
        return HttpService:JSONDecode(readfile(path))
    end)

    if not ok or type(decoded) ~= "table" then
        return false, "Macro file is invalid."
    end

    State.Macro.Events = decoded
    return true
end

-- REACTION LAB / ANIMATION WATCHER
--========================================================

local ReactionLab = {
    Seen = setmetatable({}, { __mode = "k" }),
    LastReaction = 0,
}

function ReactionLab:React(animationId, model)
    local now = os.clock()

    if now - self.LastReaction < State.Reaction.Cooldown then
        return
    end

    if State.Reaction.RuleOnly and not State.Reaction.Rules[animationId] then
        return
    end

    self.LastReaction = now

    local delayTime = State.Reaction.Rules[animationId]
        or State.Reaction.Delay

    task.delay(delayTime, function()
        if not State.Running or not State.Reaction.Enabled then
            return
        end

        if State.Reaction.Mode == "Notify" then
            card(
                "WARNING",
                "Reaction detected",
                model.Name .. " | " .. animationId,
                3
            )

        elseif State.Reaction.Mode == "Key press" and VirtualInputManager then
            pcall(function()
                VirtualInputManager:SendKeyEvent(
                    true,
                    State.Reaction.Key,
                    false,
                    game
                )

                task.wait(0.035)

                VirtualInputManager:SendKeyEvent(
                    false,
                    State.Reaction.Key,
                    false,
                    game
                )
            end)
        end
    end)
end

function ReactionLab:LearnNearest()
    local root = Character:GetRoot()

    if not root then
        return false, "Character root unavailable."
    end

    local nearest = nil
    local nearestDistance = math.huge

    for model, data in pairs(EntityESP.Objects) do
        if model.Parent
            and data.Root
            and data.Root.Parent
            and (data.Class == "MONSTER" or data.Class == "NPC") then

            local distance = (root.Position - data.Root.Position).Magnitude

            if distance < nearestDistance then
                nearestDistance = distance
                nearest = model
            end
        end
    end

    if not nearest then
        return false, "No nearby NPC/monster is tracked."
    end

    local humanoid = getEntityHumanoid(nearest)
    local animator = humanoid and humanoid:FindFirstChildOfClass("Animator")

    if not animator then
        return false, "Target has no Animator."
    end

    local tracks = animator:GetPlayingAnimationTracks()

    if #tracks == 0 then
        return false, "Target has no playing animation."
    end

    local id = tostring(tracks[1].Animation.AnimationId or "")

    if id == "" then
        return false, "Animation ID unavailable."
    end

    State.Reaction.Rules[id] = State.Reaction.Delay
    return true, id
end

local LastReactionScan = 0

connect(RunService.Heartbeat, function()
    if not State.Running or not State.Reaction.Enabled then
        return
    end

    local now = os.clock()

    if now - LastReactionScan < 0.12 then
        return
    end

    LastReactionScan = now

    local root = Character:GetRoot()

    if not root then
        return
    end

    for model, data in pairs(EntityESP.Objects) do
        if model.Parent
            and data.Root
            and data.Root.Parent
            and (data.Class == "MONSTER" or data.Class == "NPC") then

            local distance = (root.Position - data.Root.Position).Magnitude

            if distance <= State.Reaction.Radius then
                local humanoid = getEntityHumanoid(model)
                local animator = humanoid and humanoid:FindFirstChildOfClass("Animator")

                if animator then
                    for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
                        if not ReactionLab.Seen[track] then
                            ReactionLab.Seen[track] = true

                            local animation = track.Animation
                            local id = animation and tostring(animation.AnimationId or "") or ""

                            if id ~= "" then
                                ReactionLab:React(id, model)
                            end
                        end
                    end
                end
            end
        end
    end
end)

end

function ServerTools:CheckPlayer(player)
    if not player or player == LocalPlayer then
        return
    end

    if State.ServerTools.Watchlist[lower(player.Name)]
        or State.ServerTools.Watchlist[lower(player.DisplayName)] then

        card(
            "WARNING",
            "Watchlist player joined",
            State.Privacy.StreamerMode and "A watched player joined." or player.Name,
            7
        )

        if State.ServerTools.WatchMode == "Leave server" then
            pcall(function()
                LocalPlayer:Kick("Nuvem Hub watchlist: watched player joined.")
            end)
        end
    end
end

function ServerTools:Rejoin()
    if game.JobId ~= "" then
        TeleportService:TeleportToPlaceInstance(
            game.PlaceId,
            game.JobId,
            LocalPlayer
        )
    else
        TeleportService:Teleport(game.PlaceId, LocalPlayer)
    end
end

function ServerTools:Hop(lowPopulation)
    local url = string.format(
        "https://games.roblox.com/v1/games/%d/servers/Public?sortOrder=%s&limit=100",
        game.PlaceId,
        lowPopulation and "Asc" or "Desc"
    )

    local ok, decoded = pcall(function()
        return HttpService:JSONDecode(game:HttpGet(url))
    end)

    if not ok or type(decoded) ~= "table" or type(decoded.data) ~= "table" then
        return false
    end

    for _, server in ipairs(decoded.data) do
        if server.id ~= game.JobId
            and server.playing < server.maxPlayers then

            TeleportService:TeleportToPlaceInstance(
                game.PlaceId,
                server.id,
                LocalPlayer
            )

            return true
        end
    end

    return false
end

function ServerTools:SessionText()
    local ping = 0

    pcall(function()
        ping = round(LocalPlayer:GetNetworkPing() * 1000)
    end)

    return string.format(
        "PlaceId: %s\nJobId: %s\nPlayers: %d\nPing: %d ms\nFPS: %d",
        tostring(game.PlaceId),
        game.JobId ~= "" and game.JobId or "Studio / unavailable",
        #Players:GetPlayers(),
        ping,
        State.Performance.FPS
    )
end

local AutoChat = {
    Enabled = false,
    Message = "",
    Interval = 30,
    Last = 0,
}

local function sendChat(message)
    message = tostring(message or "")

    if message == "" then
        return false
    end

    local ok = pcall(function()
        local channels = TextChatService:FindFirstChild("TextChannels")
        local general = channels and channels:FindFirstChild("RBXGeneral")

        if general then
            general:SendAsync(message)
        else
            error("RBXGeneral unavailable")
        end
    end)

    return ok
end

connect(RunService.Heartbeat, function()
    if not State.Running or not AutoChat.Enabled then
        return
    end

    local now = os.clock()

    if now - AutoChat.Last >= AutoChat.Interval then
        AutoChat.Last = now
        sendChat(AutoChat.Message)
    end
end)

local HazardSandbox = {
    Originals = setmetatable({}, { __mode = "k" }),
    LastScan = 0,
    Words = {
        "kill", "lava", "death", "hazard", "spike",
        "void", "damage", "hurt", "acid", "fire",
    },
}

function HazardSandbox:Restore()
    for part, value in pairs(self.Originals) do
        if part and part.Parent then
            part.CanTouch = value
        end
    end

    table.clear(self.Originals)
end

function HazardSandbox:Update()
    if not State.Utility.HazardSandbox then
        self:Restore()
        return
    end

    local root = Character:GetRoot()

    if not root then
        return
    end

    local params = OverlapParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = { LocalPlayer.Character }
    params.MaxParts = 1800

    local parts = Workspace:GetPartBoundsInRadius(
        root.Position,
        State.Utility.HazardRadius,
        params
    )

    for _, part in ipairs(parts) do
        if part:IsA("BasePart")
            and part.CanTouch
            and containsAny(part.Name, self.Words) then

            if self.Originals[part] == nil then
                self.Originals[part] = part.CanTouch
            end

            part.CanTouch = false
        end
    end
end

connect(RunService.Heartbeat, function()
    if not State.Running then
        return
    end

    local now = os.clock()

    if State.Utility.HazardSandbox
        and now - HazardSandbox.LastScan >= 0.8 then

        HazardSandbox.LastScan = now
        HazardSandbox:Update()
    end
end)

function TargetAssist:Destroy()
    if self.Gui then
        self.Gui:Destroy()
    end

    self.Gui = nil
    self.Circle = nil
end

function TargetAssist:IsClassAllowed(class)
    local mode = State.Targeting.TargetClass

    if mode == "Players" then
        return class == "PLAYER"
    elseif mode == "Monsters" then
        return class == "MONSTER"
    elseif mode == "NPCs" then
        return class == "NPC"
    elseif mode == "Monsters + NPCs" then
        return class == "MONSTER" or class == "NPC"
    elseif mode == "Everything" then
        return class == "PLAYER"
            or class == "MONSTER"
            or class == "NPC"
            or class == "ENTITY"
    end

    return false
end

function TargetAssist:GetAimPart(model)
    if State.Targeting.TargetPart == "Head" then
        return model:FindFirstChild("Head") or getEntityRoot(model)
    elseif State.Targeting.TargetPart == "UpperTorso" then
        return model:FindFirstChild("UpperTorso")
            or model:FindFirstChild("Torso")
            or getEntityRoot(model)
    end

    return getEntityRoot(model)
end

function TargetAssist:HasLineOfSight(targetModel, targetPart)
    if not State.Targeting.WallCheck then
        return true
    end

    Camera = Workspace.CurrentCamera

    if not Camera then
        return false
    end

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {
        LocalPlayer.Character,
        targetModel,
    }

    local direction = targetPart.Position - Camera.CFrame.Position
    local result = Workspace:Raycast(
        Camera.CFrame.Position,
        direction,
        params
    )

    return result == nil
end

function TargetAssist:FindTarget(center)
    Camera = Workspace.CurrentCamera

    if not Camera then
        return nil
    end

    local best = nil
    local bestDistance = State.Targeting.FOV

    for model, data in pairs(EntityESP.Objects) do
        if model.Parent
            and data.Root
            and data.Root.Parent
            and self:IsClassAllowed(data.Class) then

            local part = self:GetAimPart(model)

            if part and part.Parent then
                local point, visible = Camera:WorldToViewportPoint(part.Position)

                if visible and point.Z > 0 then
                    local screenDistance = (
                        Vector2.new(point.X, point.Y) - center
                    ).Magnitude

                    if screenDistance < bestDistance
                        and self:HasLineOfSight(model, part) then

                        bestDistance = screenDistance
                        best = part
                    end
                end
            end
        end
    end

    return best
end

connect(RunService.RenderStepped, function(dt)
    if not State.Running then
        return
    end

    if State.Targeting.Enabled then
        TargetAssist:EnsureGui()
    end

    if TargetAssist.Circle then
        local center

        Camera = Workspace.CurrentCamera

        if Library.IsMobile and Camera then
            local size = Camera.ViewportSize
            center = Vector2.new(size.X / 2, size.Y / 2)
        else
            center = UIS:GetMouseLocation()
        end

        TargetAssist.Circle.Visible = State.Targeting.Enabled and State.Targeting.ShowFOV
        TargetAssist.Circle.Position = UDim2.fromOffset(center.X, center.Y)
        TargetAssist.Circle.Size = UDim2.fromOffset(
            State.Targeting.FOV * 2,
            State.Targeting.FOV * 2
        )

        if not State.Targeting.Enabled
            or State.Freecam.Enabled
            or (
                State.Targeting.HoldRightMouse
                and not Library.IsMobile
                and not UIS:IsMouseButtonPressed(Enum.UserInputType.MouseButton2)
            ) then
            return
        end

        local target = TargetAssist:FindTarget(center)

        if target and Camera then
            local desired = CFrame.lookAt(
                Camera.CFrame.Position,
                target.Position
            )

            local alpha = 1 - math.exp(
                -math.max(State.Targeting.Smoothness, 0.01) * 28 * dt
            )

            Camera.CFrame = Camera.CFrame:Lerp(desired, clamp(alpha, 0, 1))
        end
    end
end)

local RuntimeConsole = {
    Lines = {},
    MaxLines = 120,
}

function RuntimeConsole:Add(message, messageType)
    local prefix = tostring(messageType or "Output")
    local line = "[" .. prefix .. "] " .. tostring(message)

    table.insert(self.Lines, line)

    while #self.Lines > self.MaxLines do
        table.remove(self.Lines, 1)
    end
end

function RuntimeConsole:GetRecent(count)
    count = count or 25

    local startIndex = math.max(1, #self.Lines - count + 1)
    local output = {}

    for index = startIndex, #self.Lines do
        table.insert(output, self.Lines[index])
    end

    return table.concat(output, "\n")
end

connect(LogService.MessageOut, function(message, messageType)
    RuntimeConsole:Add(message, messageType.Name)
end)

        if not nearest then
            card(
                "WARNING",
                "Access Finder",

        Interactables:Pin(nearest.Object)

QuickBox:AddButton({
    Text = "Find item for nearest access",

    Func = function()
        task.spawn(function()
            Interactables:ScanLocal()

            local access =
                Interactables:
                GetNearestRequiredAccess()

            if not access then
                card(
                    "WARNING",
                    "Requirement Finder",
                    "No required access point indexed nearby.",
                    5
                )
                return
            end

            local requirement =
                access.Data.Info.Access
                and access.Data.Info.Access.Item

            if not requirement then
                card(
                    "INFO",
                    access.Object.Name,
                    "This access point does not expose a required item.",
                    5
                )
                return
            end

            local n =
                card(
                    "SCAN",
                    "Searching for " .. requirement,
                    "Checking indexed objects...",
                    12
                )

            local match =
                Interactables:
                FindBestItem(
                    requirement
                )

            if match then
                Interactables:Pin(match.Object)

                local text =
                    string.format(
                        "%s\n%d studs away\n%d%% name match",
                        match.Object.Name,
                        round(match.Distance),
                        round(match.Match)
                    )

                if n
                    and n.ChangeTitle
                    and n.ChangeDescription then

                    n:ChangeTitle("Item found")
                    n:ChangeDescription(text)
                else
                    card(
                        "SUCCESS",
                        "Item found",
                        text,
                        7
                    )
                end
            else
                if n
                    and n.ChangeTitle
                    and n.ChangeDescription then

                    n:ChangeTitle("Item not found")
                    n:ChangeDescription(
                        "No client-visible match found for: "
                        .. requirement
                    )
                end
            end
        end)
    end,
})

FinderBox:AddButton({
    Text = "Find & pin target",

    Func = function()
        task.spawn(function()
            if not SearchValue
                or SearchValue == "" then
                return
            end

            local n =
                card(
                    "SCAN",
                    "Searching: " .. SearchValue,
                    "Checking indexed objects...",
                    12
                )

            local result =
                Interactables:
                FindByName(
                    SearchValue
                )

            if not result then
                if n and n.ChangeDescription then
                    n:ChangeDescription(
                        "Not indexed. Running one global name search..."
                    )
                end

                result =
                    Interactables:
                    SearchGlobal(
                        SearchValue
                    )
            end

            if result then
                Interactables:
                    Pin(
                        result.Object
                    )

                local description =
                    string.format(
                        "%s\n%s | %d studs | %d%% match",
                        result.Object.Name,
                        prettyClass(
                            result.Data.Info.Class
                        ),
                        round(result.Distance),
                        round(result.Match)
                    )

                if n
                    and n.ChangeTitle
                    and n.ChangeDescription then

                    n:ChangeTitle("Target found")
                    n:ChangeDescription(description)
                end
            else
                if n
                    and n.ChangeTitle
                    and n.ChangeDescription then

                    n:ChangeTitle("No match found")
                    n:ChangeDescription(
                        "No client-visible object matched: "
                        .. SearchValue
                    )
                end
            end
        end)
    end,
})

    local okSave, SaveManager =
        pcall(
            loadCached,
            "Obsidian_SaveManager.lua",
            "addons/SaveManager.lua"
        )

    if okTheme and okSave then
        ThemeManager:SetLibrary(Library)
        SaveManager:SetLibrary(Library)

        SaveManager:IgnoreThemeSettings()
        SaveManager:SetIgnoreIndexes({
            "MenuKeybind",
            "TargetSearch",
        })

        ThemeManager:SetFolder("NuvemHub")
        SaveManager:SetFolder("NuvemHub")
        SaveManager:SetSubFolder(
            tostring(game.PlaceId)
        )

        ThemeManager:ApplyToTab(
            Tabs.Settings,
            "palette"
        )

        SaveManager:BuildConfigSection(
            Tabs.Settings,
            "save"
        )

        task.defer(function()
            pcall(function()
                SaveManager:
                    LoadAutoloadConfig()
            end)
        end)
    end
end

--========================================================
-- CLEANUP
--========================================================

local Cleaned = false

local function cleanup()
    if Cleaned then
        return
    end

    Cleaned = true
    State.Running = false

    cleanupUniversalTools()

    EntityESP:Clear()
    Interactables:Clear()

    restoreCollision()
    Character:Restore()

    Camera = Workspace.CurrentCamera

    if Camera then
        Camera.FieldOfView = OriginalCamera.FOV
    end

    LocalPlayer.CameraMinZoomDistance =
        OriginalCamera.MinZoom

    LocalPlayer.CameraMaxZoomDistance =
        OriginalCamera.MaxZoom

    Lighting.Brightness =
        OriginalLighting.Brightness

    Lighting.ClockTime =
        OriginalLighting.ClockTime

    Lighting.FogStart =
        OriginalLighting.FogStart

    Lighting.FogEnd =
        OriginalLighting.FogEnd

    Lighting.Ambient =
        OriginalLighting.Ambient

    Lighting.OutdoorAmbient =
        OriginalLighting.OutdoorAmbient

    Lighting.GlobalShadows =
        OriginalLighting.GlobalShadows

    Workspace.Gravity =
        OriginalWorld.Gravity

    TargetHUD:SetVisible(false)
    PerformanceHUD:SetVisible(false)
    ThreatHUD:SetVisible(false)

    disconnectAll()

    _G.__NUVEM_HUB = nil
end

Library:OnUnload(cleanup)

local function unload()
    if not Cleaned then
        Library:Unload()
    end
end

MaintenanceBox:AddButton({
    Text = "Unload Nuvem Hub",
    Risky = true,
    DoubleClick = true,

    Func = unload,
})
